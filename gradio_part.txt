# TORILO ACADEMY CHATBOT LOGIC
# ============================================================

def chat_with_student(message, history, chatbot_state):
    """
    Process a student's question using the existing
    Torilo Academy chatbot engine.
    """

    if not message or not message.strip():
        return "Please enter a question.", chatbot_state

    # Keep chatbot memory isolated for each browser session.
    if chatbot_state is None:
        chatbot_state = ToriloAcademyChatbot(
            top_k=3,
            max_memory_turns=3
        )

    try:
        result = chatbot_state.answer(message.strip())

        answer = result["answer"]
        source = result["source"]

        if isinstance(source, list):
            source_text = ", ".join(source)
        else:
            source_text = str(source)

        response = (
            f"{answer}\n\n"
            f"**Source:** {source_text}"
        )

        return response, chatbot_state

    except Exception as error:
        print(f"Chatbot error: {error}")

        return (
            "Sorry, I encountered an error while "
            "processing your question.",
            chatbot_state
        )
